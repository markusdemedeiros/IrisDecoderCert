ff_vector_fmul_avx:
    lea rcx, [ecx*4+-64] ;; aw=w32 ow=w64
    .nops 7 ;; aw=w64 ow=w64
ff_vector_fmul_avx.loop:
    vmovaps ymm0, YMMWORD PTR [rsi+rcx*1+0] ;; aw=w64 ow=w256
    vmovaps ymm1, YMMWORD PTR [rcx+rsi*1+32] ;; aw=w64 ow=w256
    vmulps ymm0, ymm0, YMMWORD PTR [rdx+rcx*1+0] ;; aw=w64 ow=w256
    vmulps ymm1, ymm1, YMMWORD PTR [rcx+rdx*1+32] ;; aw=w64 ow=w256
    vmovaps YMMWORD PTR [rdi+rcx*1+0], ymm0 ;; aw=w64 ow=w256
    vmovaps YMMWORD PTR [rcx+rdi*1+32], ymm1 ;; aw=w64 ow=w256
    sub rcx, 64 ;; aw=w64 ow=w64
    jge ff_vector_fmul_avx.loop ;; aw=w64 ow=w64
ff_vector_fmul_avx.done:
    vzeroupper ;; aw=w64 ow=w128
    ret ;; aw=w64 ow=w64
